\section{Background \& Literature Survey}

\begin{frame}{IoT Authentication Architectures}
    \begin{columns}
        \begin{column}{0.46\textwidth}
            Authentication in IoT typically involves three entities:
            \begin{itemize}
                \item User / device $U_i$
                \item Sensor / peer node $S_j$
                \item Gateway node (GWN)
            \end{itemize}
            \vspace{0.3cm}
            Four prevalent communication architectures (a)--(d) differ in how the
            gateway mediates the exchange. Our work targets the \textbf{multihop}
            device--gateway path with a \textbf{distributed} authenticator.
        \end{column}

\begin{frame}{Related Work}
    \begin{itemize}
        \item \textbf{Centralized} schemes (Sheik'23, He'23, Roy'25): high computation/communication
        overhead in multihop networks.
        \item \textbf{Distributed} schemes (RAAF-MEC'25): skewed load on particular fog/edge servers;
        ignore multihop IoT.
        \item \textbf{LightSAE'23}: parents authenticate children $\Rightarrow$ skewed overhead.
        \item \textbf{PUF-based P2P} (Li'20, Zheng'22): avoid long-term key storage, but still skewed;
        ECC operations make them heavy.
        \item \textbf{LAAKA'24}, \textbf{Zhou \emph{et al.}'24}: provide anonymity, but rely on
        \textbf{computationally intensive} operations $\Rightarrow$ far higher overhead than DAuth.
        \item \textbf{DAuth'26 (base scheme, Das \& Khatua)}: \textbf{decoupled} distributed PUF-based
        scheme with a hash accumulator $\Rightarrow$ lightweight, but
        \textbf{no anonymity} and \textbf{no desync recovery}.
    \end{itemize}
\end{frame}

\section{Motivation \& Contributions}

\begin{frame}{Motivation: Two Gaps in DAuth}
    DAuth is the closest match in the comparison, yet it leaves \textbf{two critical
    security gaps} unaddressed:

    \vspace{0.3cm}
    \begin{enumerate}
        \item \textbf{No anonymity.} The real device identity $ID_D$ is sent in
        \textbf{plaintext on the open channel in every session}.
        \begin{itemize}
            \item An adversary with radio access can \textbf{track a device across sessions}.
            \item Direct violation of modern privacy requirements.
        \end{itemize}

        \vspace{0.25cm}
        \item \textbf{No desynchronisation recovery.} AS and device keep only a
        \textbf{single session state}.
        \begin{itemize}
            \item A single lost packet during key delivery \textbf{desynchronises} their states.
            \item All future authentications fail until \textbf{full re-enrolment} --
            an energy-costly operation for a battery-powered node.
        \end{itemize}
    \end{enumerate}
\end{frame}

\begin{frame}{Our Contributions}
    We extend DAuth with \textbf{two lightweight mechanisms} that close both gaps:

    \vspace{0.3cm}
    \begin{block}{1. Per-session Pseudonym Rotation (Anonymity)}
        Real identity $ID_D$ never leaves the secure enrolment channel.
        Open-channel messages carry a rotating pseudonym
        $\mathit{PID} = H(ID_D \,\|\, m_{\mathrm{curr}})$, refreshed after every key exchange
        $\Rightarrow$ successive sessions are \textbf{unlinkable}.
    \end{block}

    \begin{block}{2. Dual-State Synchronisation (Desync Recovery)}
        AS and gateway store \emph{both} $(m_{\mathrm{curr}}, \mathit{PID}_{\mathrm{curr}})$
        and $(m_{\mathrm{old}}, \mathit{PID}_{\mathrm{old}})$ $\Rightarrow$ recover from a single
        packet loss \textbf{without re-enrolment}.
    \end{block}

    \vspace{0.15cm}
    \begin{itemize}
        \item Formally verified in \textbf{ProVerif} (Dolev--Yao model).
        \item Evaluated via \textbf{theoretical}, \textbf{COOJA/Contiki-NG simulation}, and \textbf{hardware (RPi 4B)} analysis.
    \end{itemize}
\end{frame}
