\section{Background \& Literature Survey}

\begin{frame}{Related Work}
    \scriptsize
    \begin{block}{Centralized Authentication}
        \begin{itemize}
            \item \textbf{Sheik}~\cite{sheik}, \textbf{Roy}~\cite{roy}: single authenticator
            $\Rightarrow$ bottleneck and high computation/communication overhead in multihop networks.
            \item \textbf{He}~\cite{he}: lightweight hash/XOR, but tied to a \textbf{centralized server}
            $\Rightarrow$ forfeits load distribution.
            \item \textbf{LAAKA}~\cite{laaka}: anonymous (hash/XOR), but \textbf{gateway/fog-coupled},
            \textbf{no PUF}, larger message payloads.
            \item \textbf{Zhou \emph{et al.}}~\cite{zhou}: anonymous PUF-based, but \textbf{gateway-coupled}
            and \textbf{no forward secrecy} (no DH key exchange).
        \end{itemize}
    \end{block}
    \begin{block}{Distributed Authentication}
        \begin{itemize}
            \item \textbf{RAAF-MEC}~\cite{raaf}: \textbf{skewed load} on particular fog/edge servers;
            ignores multihop IoT.
            \item \textbf{LightSAE}~\cite{lightsae}: parents authenticate children $\Rightarrow$
            \textbf{overloaded parents}, idle leaves (skewed load).
            \item \textbf{Li}~\cite{li}, \textbf{Zheng}~\cite{zheng} (PUF P2P): no long-term key
            storage, but still \textbf{skewed} and \textbf{ECC-heavy}.
            \item \textbf{DAuth}~\cite{dauth}: \textbf{decoupled} PUF + hash accumulator $\Rightarrow$
            lightweight, but \textbf{no anonymity} and \textbf{no desync recovery}.
        \end{itemize}
    \end{block}
\end{frame}
