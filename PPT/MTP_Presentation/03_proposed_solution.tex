\section{Pre-Requisites}

\begin{frame}{System Model \& Assumptions}
    \scriptsize
    \setlength{\itemsep}{0pt}
    \begin{columns}[T]
        \begin{column}{0.55\textwidth}
            \textbf{Entities}
            \begin{itemize}\setlength{\itemsep}{1pt}
                \item \textbf{$D$} -- newly joined device (PUF, identity $ID_D$).
                \item \textbf{$AS$} -- \emph{any} capable node pre-selected by the gateway (parent or non-parent).
                \item \textbf{$GW$} -- trusted gateway / root (registration authority).
            \end{itemize}
            \vspace{0.1cm}
            \textbf{Assumptions}
            \begin{itemize}\setlength{\itemsep}{1pt}
                \item $GW$ shares a long-term key $K_{GW\text{-}AS}$ with each AS.
                \item $D$ knows the address and identity $ID_{AS}$ of its AS.
                \item Nodes can do hash, AES, XOR, AND, concat, PUF eval.
            \end{itemize}
        \end{column}
        \begin{column}{0.45\textwidth}
            \begin{block}{Three protocol phases}
                \begin{enumerate}\setlength{\itemsep}{1pt}
                    \item \textbf{Enrolment} {\scriptsize (once, secure channel)}
                    \item \textbf{Authentication} {\scriptsize (open channel)}
                    \item \textbf{Key Exchange} {\scriptsize (open channel)}
                \end{enumerate}
            \end{block}
            \begin{exampleblock}{Decoupling}
                Authentication is \emph{decoupled} from network joining -- no parent is overwhelmed by its children.
            \end{exampleblock}
        \end{column}
    \end{columns}
    \vspace{0.15cm}
    \begin{block}{Threat Model (Dolev--Yao)}
        \begin{itemize}\setlength{\itemsep}{1pt}
            \item \textbf{Channels:} enrolment \textbf{secure \& encrypted}; public channels \textbf{insecure}, fully adversary-controlled.
            \item \textbf{Adversary \emph{can}} intercept, replay, modify, inject public messages; \textbf{\emph{cannot}} break hashes, read/clone the PUF, or access device secrets ($y_D,\, m_{\mathrm{curr}}$).
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{Building Blocks -- PUF \& Hash Accumulator}
    \vspace{0.4cm}
    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.48\textwidth}
            \begin{block}{Physically Unclonable Function (PUF)}
                \footnotesize
                A hardware ``fingerprint'': manufacturing variations map a
                challenge $c$ to a unique, unclonable response $R=\mathsf{PUF}(c)$.
            \end{block}
            \vspace{0.4cm}
            \textbf{Significance}
            \begin{itemize}\footnotesize
                \item \textbf{No stored secret:} $R_D$ is \emph{regenerated} on demand,
                never kept in memory $\Rightarrow$ nothing to steal if a node is captured.
                \item \textbf{Unclonable:} cannot be copied even with the chip design
                $\Rightarrow$ device cannot be impersonated.
            \end{itemize}
        \end{column}
        \hfill
        \begin{column}{0.48\textwidth}
            \begin{block}{Hash-based One-Way Accumulator}
                \footnotesize
                A single value $T_{\mathsf{Acc}}$ that compactly represents the whole
                membership set: $T_{\mathsf{Acc}} \mathrel{\&}{=} H(y_D)$ on each enrolment.
            \end{block}
            \vspace{0.4cm}
            \textbf{Significance}
            \begin{itemize}\footnotesize
                \item \textbf{Compact membership:} one fixed-size value stands in for
                \emph{all} enrolled devices' secrets.
                \item \textbf{One-way:} individual secrets $y_D$ cannot be recovered
                from $T_{\mathsf{Acc}}$.
                \item \textbf{Cheap test:} a legitimate device passes
                $T_{\mathsf{Acc}} \mathrel{\&} H(y_D) = T_{\mathsf{Acc}}$ -- verification
                is a single bitwise AND.
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

\section{Motivation \& Contributions}

\begin{frame}{Motivation: Two Gaps in DAuth}
    DAuth leaves \textbf{two critical security gaps} unaddressed:

    \vspace{0.3cm}
    \begin{enumerate}
        \item \textbf{No anonymity.} The real device identity $ID_D$ is sent in
        \textbf{plaintext on the open channel in every session}.
        \begin{itemize}
            \item An adversary with radio access can \textbf{track a device across sessions}.
            \item Direct violation of modern privacy requirements.
        \end{itemize}

        \vspace{0.25cm}
        \item \textbf{No desynchronisation recovery.} AS and device keep only a
        \textbf{single session state}.
        \begin{itemize}
            \item A single lost packet during key delivery \textbf{desynchronises} their states.
            \item All future authentications fail until \textbf{full re-enrolment} --
            an energy-costly operation for a battery-powered node.
        \end{itemize}
    \end{enumerate}
\end{frame}

\begin{frame}{Our Contributions}
    We extend DAuth with \textbf{two lightweight mechanisms} that close both gaps:

    \vspace{0.3cm}
    \begin{block}{1. Per-session Pseudonym Rotation (Anonymity)}
        Real identity $ID_D$ never leaves the secure enrolment channel.
        Open-channel messages carry a rotating pseudonym
        $\mathit{PID} = H(ID_D \,\|\, m_{\mathrm{curr}})$, refreshed after every key exchange
        $\Rightarrow$ successive sessions are \textbf{unlinkable}.
    \end{block}

    \begin{block}{2. Dual-State Synchronisation (Desync Recovery)}
        The AS stores \emph{both} $(m_{\mathrm{curr}}, \mathit{PID}_{\mathrm{curr}})$
        and $(m_{\mathrm{old}}, \mathit{PID}_{\mathrm{old}})$ $\Rightarrow$ recover from a single
        packet loss \textbf{without re-enrolment}.
    \end{block}

    \vspace{0.15cm}
    \begin{itemize}
        \item Formally verified in \textbf{ProVerif} (Dolev--Yao model).
        \item Evaluated via \textbf{theoretical}, \textbf{COOJA/Contiki-NG simulation}, and \textbf{hardware (RPi 4B)} analysis.
    \end{itemize}
\end{frame}

\section{Proposed Scheme}

\begin{frame}{Key Idea 1 -- Per-Session Pseudonym Rotation}
    \begin{columns}
        \begin{column}{0.55\textwidth}
            \begin{itemize}
                \item Real $ID_D$ is sent \textbf{only once}, over the secure enrolment channel.
                \item Every open-channel message uses a pseudonym:
                $$\mathit{PID} = H(ID_D \,\|\, m_{\mathrm{curr}})$$
                \item After each key exchange, $m_{\mathrm{curr}} \!\rightarrow\! m_{\mathrm{new}}$,
                so $\mathit{PID}$ rotates.
                \item $H$ is one-way and $m_{\mathrm{curr}}$ is private $\Rightarrow$
                consecutive pseudonyms are \textbf{computationally unlinkable}.
            \end{itemize}
        \end{column}
        \begin{column}{0.45\textwidth}
            \begin{block}{Effect}
                \centering
                \footnotesize
                Session 1: $\mathit{PID}_1 = H(ID_D \| m_1)$ \\[0.2cm]
                Session 2: $\mathit{PID}_2 = H(ID_D \| m_2)$ \\[0.2cm]
                $\cdots$ \\[0.2cm]
                \textbf{An observer cannot link}
                $\mathit{PID}_1, \mathit{PID}_2, \ldots$ to the same device.
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Phase 1 -- Node Enrolment (Secure Channel)}
    \begin{columns}
        \begin{column}{0.46\textwidth}
            \footnotesize
            \begin{itemize}
                \item $D \to AS$: $\langle \mathsf{Reg\_Req},\, ID_D\rangle$ -- registration request.
                \item $AS$ issues challenge $c_D$ and session random $m_D$.
                \item $D$ computes PUF response $R_D=\mathsf{PUF}_D(c_D)$, picks secret $y_D$,
                returns $(y_D, R_D, c_{AS\text{-}D})$.
                \item $AS$ folds $H(y_D)$ into accumulator $T_{\mathsf{Acc}}$ (bitwise AND),
                stores $\Phi_{AS\text{-}D}=R_{AS\text{-}D}\oplus R_D$.
                \item \textbf{Pseudonym init:} $AS$ sets $\mathit{PID}_{\mathrm{curr}}=H(ID_D\|m_{\mathrm{curr}})$,
                $\mathit{PID}_{\mathrm{old}}=\varepsilon$ -- the initial \textbf{dual state}.
            \end{itemize}
        \end{column}
        \begin{column}{0.54\textwidth}
            \centering
            \includegraphics[width=\textwidth]{fig_enrollment.jpg}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Phase 2 -- Node Authentication (Open Channel)}
    \begin{columns}
        \begin{column}{0.46\textwidth}
            \footnotesize
            \begin{itemize}
                \item $D$ regenerates $R_D$ from $c_D$ and computes $\mathit{PID}=H(ID_D\|m_{\mathrm{curr}})$.
                \item Sends masked tag
                $Y_{AS\text{-}D}^H = H(y_D)\oplus H(R_D\|m_{\mathrm{curr}}\|\mathit{PID}\|ts_1)$.
                \item Message: $(\mathit{PID},\, Y_{AS\text{-}D}^H,\, ts_1)$ -- \textbf{no $ID_D$}.
                \item $AS$ does a \textbf{dual-state lookup} ($\mathit{PID}_{\mathrm{curr}}$ or $\mathit{PID}_{\mathrm{old}}$),
                checks $ts_1$ freshness.
                \item $AS$ recovers $R_D'$, unmasks $Y_D^{H'}$, verifies via the
                \textbf{accumulator AND-check} ($T_{\mathsf{Acc}}^{\mathsf{new}}\!=\!T_{\mathsf{Acc}}$).
            \end{itemize}
        \end{column}
        \begin{column}{0.54\textwidth}
            \centering
            \includegraphics[width=\textwidth]{fig_auth.jpg}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Phase 3 -- Session Key Exchange (Open Channel)}
    \begin{columns}
        \begin{column}{0.46\textwidth}
            \footnotesize
            \begin{itemize}
                \item $AS$ draws fresh $n_1$, sets $m_{\mathrm{new}}=H(n_1)$.
                \item Session key $K_{GW\text{-}D}=H(R_D'\|m_{\mathrm{new}})$.
                \item New pseudonym $\mathit{PID}_{\mathrm{new}}=H(ID_D\|m_{\mathrm{new}})$.
                \item $AS \to GW$: $\mathsf{auth\_token}_D$ (encrypted under $K_{GW\text{-}AS}$, carries $K_{GW\text{-}D}$ + $\mathit{PID}_{\mathrm{new}}$).
                \item $AS \to D$: masked $m_H = m_{\mathrm{new}}\oplus H(\cdots)$ + $ts_2$.
                \item $D$ recovers $m_{\mathrm{new}}$, derives the \textbf{same} $K_{GW\text{-}D}$, rotates $\mathit{PID}$.
                \item $AS$ advances dual state (old $\leftarrow$ curr, curr $\leftarrow$ new); $GW$ just updates the session to $\mathit{PID}_{\mathrm{new}}$.
            \end{itemize}
        \end{column}
        \begin{column}{0.54\textwidth}
            \centering
            \includegraphics[width=\textwidth]{fig_keyex.jpg}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Key Idea 2 -- Dual-State Desynchronisation Recovery}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \textbf{The problem}
            \begin{itemize}
                \item In Phase 3, $AS$ sends $(m_H, ts_2)$ to $D$.
                \item If this packet is \textbf{lost}, $AS$ has already advanced to $m_{\mathrm{new}}$,
                but $D$ keeps stale $m_{\mathrm{curr}}$.
                \item $D$'s next $\mathit{PID}$ matches nothing $\Rightarrow$ authentication fails.
            \end{itemize}
            \vspace{0.2cm}
            \textbf{Base scheme (DAuth):} single state $\Rightarrow$ \textbf{full re-enrolment}.
        \end{column}
        \begin{column}{0.5\textwidth}
            \begin{block}{Our fix: dual state}
                Only $AS$ keeps \emph{both}
                $(m_{\mathrm{curr}},\mathit{PID}_{\mathrm{curr}})$ and
                $(m_{\mathrm{old}},\mathit{PID}_{\mathrm{old}})$.
            \end{block}
            \begin{itemize}\footnotesize
                \item Stale $D$ arrives with $\mathit{PID}=\mathit{PID}_{\mathrm{old}}$.
                \item $AS$ recognises it via dual-state lookup, verifies using $m_{\mathrm{old}}$,
                and re-synchronises -- \textbf{no re-enrolment}.
                \item $D$'s behaviour is \emph{identical}; only $AS$ adapts.
            \end{itemize}
            \begin{alertblock}{Trade-off}
                \footnotesize Tolerates a single consecutive loss; two slots balance storage vs.\ recovery depth.
            \end{alertblock}
        \end{column}
    \end{columns}
\end{frame}
